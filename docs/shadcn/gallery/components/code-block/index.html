Code Block

A framed code listing with a header, actions, a copy button, line numbers and diff lines.

Demo

main.go 
package main

import "fmt"

func main() {
	fmt.Println("Hello, go-mx")
}
html.DivClass("w-full max-w-md",
	codeTheme(),
	shadcn.CodeBlock("", "",
		shadcn.CodeBlockHeader(
			shadcn.CodeBlockTitle(iconFileText(), "main.go"),
			shadcn.CodeBlockActions(shadcn.CodeBlockCopyButton(codeBlockExampleSource)),
		),
		shadcn.CodeBlockContent(codeHighlighter.Components(highlight.TokenizeGo(codeBlockExampleSource))),
	),
)

Highlighted line numbers

main.go 
package main
import "fmt"
func main() {	fmt.Println("Hello, go-mx")}
content := []any{shadcn.CodeBlockLineNumbers}
for i, line := range codeHighlighter.Lines(highlight.TokenizeGo(codeBlockExampleSource)) {
	variant := shadcn.CodeBlockLineDefault
	if i == 5 {
		variant = shadcn.CodeBlockLineHighlighted
	}
	content = append(content, shadcn.CodeBlockLine(variant, line))
}
return html.DivClass("w-full max-w-md",
	codeTheme(),
	shadcn.CodeBlock("", "",
		shadcn.CodeBlockHeader(
			shadcn.CodeBlockTitle(iconFileText(), "main.go"),
			shadcn.CodeBlockActions(shadcn.CodeBlockCopyButton("")),
		),
		shadcn.CodeBlockContent(content...),
	),
)
